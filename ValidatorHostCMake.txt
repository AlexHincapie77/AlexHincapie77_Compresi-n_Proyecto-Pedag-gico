cmake_minimum_required(VERSION 3.22)

project(ValidatorHost LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT TARGET juce::juce_audio_processors)
    # Reutilizar el JUCE ya descargado por el proyecto del plugin
    include(FetchContent)
    FetchContent_Declare(JUCE
        GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
        GIT_TAG        8.0.4
        GIT_SHALLOW    TRUE
        SOURCE_DIR     ${CMAKE_CURRENT_SOURCE_DIR}/build/_deps/juce-src
        BINARY_DIR     ${CMAKE_CURRENT_SOURCE_DIR}/build/_deps/juce-build
    )
    FetchContent_MakeAvailable(JUCE)
endif()

juce_add_console_app(validate_all
    NEEDS_CURL FALSE)

target_sources(validate_all PRIVATE
    /home/z/my-project/scripts/validate_all.cpp)

target_compile_definitions(validate_all PRIVATE
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_DISPLAY_SPLASH_SCREEN=0
    JUCE_REPORT_APP_USAGE=0
    JUCE_VST3_CAN_REPLACE_VST2=0
)

target_link_libraries(validate_all PRIVATE
    juce::juce_audio_utils
    juce::juce_audio_processors
    juce::juce_dsp
)

target_link_libraries(validate_all PUBLIC
    juce::juce_recommended_config_flags
    juce::juce_recommended_warning_flags
)
